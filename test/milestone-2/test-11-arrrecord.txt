program ArrayRecordTest;
type
  Person == record
    name: string;
    age: integer
  end;
var
  arr: array[1..10] of integer;
  people: array[1..5] of Person;
  p: Person;
begin
  arr[1] := 10;
  arr[5] := arr[1] + 20;
  p.name := 'John';
  p.age := 25;
  people[1].name := 'Alice';
  people[2].age := 30;
end.
