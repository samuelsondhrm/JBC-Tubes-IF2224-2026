program ForLoopType;
var
  i, total: integer;
  r: real;
begin
  total := 0;
  r := 1.5;

  for i := 1 to 5 do
  begin
    total := total + i;
  end;

  for i := 5 downto 1 do
  begin
    total := total + i;
  end;

  for r := 1 to 3 do     { ERROR: for loop variable must be integer }
  begin
    total := total + 1;
  end;

  for i := r to 3 do     { ERROR: lower bound must be integer }
  begin
    total := total + 1;
  end;

  for i := 1 to r do     { ERROR: upper bound must be integer }
  begin
    total := total + 1;
  end;

  total := total + 1;
end.
