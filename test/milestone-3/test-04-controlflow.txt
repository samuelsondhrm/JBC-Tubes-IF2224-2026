program ControlFlow;
var
  x: integer;
  y: real;
  b: boolean;
begin
  x := 5;
  if x == 5 then
    x := 10;
  if x > 0 then
    y := 3.14
  else
    y := 0.0;
  while x > 0 do
    x := x - 1;
  for x := 1 to 10 do
    y := y + 1.0;
  for x := 10 downto 1 do
    y := y - 1.0;
  repeat
    x := x + 1
  until x == 20;
  case x of
    1: y := 1.0;
    2: y := 2.0
  end;
  b := x > 0;
  b := not b;
  b := (x == 1) and (y == 2.0);
  b := (x == 1) or (y == 2.0);
end.
