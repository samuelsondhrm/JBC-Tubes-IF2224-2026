program ConditionType;
var
  i: integer;
  r: real;
  ok: boolean;
begin
  i := 0;
  r := 1.5;
  ok := true;

  if ok then
  begin
    i := i + 1;
  end;

  if i < 10 then
  begin
    ok := false;
  end;

  while ok do
  begin
    ok := false;
  end;

  repeat
    i := i + 1;
  until i >= 2;

  if i then             { ERROR: if condition must be boolean }
  begin
    i := 1;
  end;

  while r do            { ERROR: while condition must be boolean }
  begin
    r := r + 1.0;
  end;

  repeat
    i := i + 1;
  until r;              { ERROR: repeat condition must be boolean }
end.
