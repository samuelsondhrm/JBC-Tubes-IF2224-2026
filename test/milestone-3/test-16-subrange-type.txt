program SubrangeType;
type
  Small == 1..5;
  Bigger == 1..10;
  BadReal == 1.5..3.5;   { ERROR: real-based subrange is invalid }
  Backwards == 10..1;    { ERROR: lower bound greater than upper bound }
var
  s: Small;
  b: Bigger;
  i: integer;
  r: real;
begin
  s := 1;
  s := 5;
  b := s;
  i := s;
  s := i;                { allowed only if value is within range }

  s := 6;                { ERROR: outside Small range }
  s := 0;                { ERROR: outside Small range }
  s := r;                { ERROR: subrange := real }
  r := s;                { valid widening via integer-based subrange }
end.
