program FunctionReturnType;

function GoodInteger(a: integer): integer;
begin
  GoodInteger := a + 1;
end;

function GoodReal(a: integer): real;
begin
  GoodReal := a / 2;
end;

function BadIntegerReturn(a: integer): integer;
begin
  BadIntegerReturn := a / 2;      { ERROR: integer function assigned real }
end;

function BadBooleanReturn(a: integer): boolean;
begin
  BadBooleanReturn := a + 1;      { ERROR: boolean function assigned integer }
end;

begin
  GoodInteger(1);
  GoodReal(2);
end.
