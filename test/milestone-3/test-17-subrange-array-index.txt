program SubrangeArrayIndex;
type
  Index == 1..3;
var
  idx: Index;
  arr: array[Index] of integer;
  i: integer;
  r: real;
  ok: boolean;
begin
  idx := 1;
  i := 2;
  r := 1.5;
  ok := false;

  arr[idx] := 10;
  arr[1] := arr[idx] + i;
  arr[3] := 30;

  arr[4] := 40;          { ERROR: index outside Index range }
  arr[0] := 50;          { ERROR: index outside Index range }
  arr[r] := 60;          { ERROR: real index }
  arr[idx] := r;         { ERROR: integer array element assigned real }
  arr[idx] := ok;        { ERROR: integer array element assigned boolean }
end.
