program ArrayRecordCombo;
type
  Item == record
    id: integer;
    weight: real;
    active: boolean
  end;
var
  items: array[1..3] of Item;
  one: Item;
  i: integer;
begin
  i := 1;
  items[1].id := 10;
  items[1].weight := 2.5;
  items[i].active := true;
  one.id := items[1].id;
  one.weight := items[1].weight;
  one.active := items[1].active;

  items[1].id := 3.14;       { ERROR: integer field assigned real }
  items[2].missing := 1;     { ERROR: field missing not found }
  items[3] := 5;             { ERROR: record element assigned integer }
  one := 7;                  { ERROR: record assigned integer }
end.
