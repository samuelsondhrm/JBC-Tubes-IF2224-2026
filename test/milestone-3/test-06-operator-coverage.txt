program OperatorCoverage;
var
  i, j, k: integer;
  r, s: real;
  flag, other: boolean;
  ch1, ch2: char;
  text1, text2: string;
begin
  i := 10 + 2;
  j := i - 3;
  k := i * j;
  r := i / 2;
  i := j div 2;
  k := j mod 3;
  s := r + i;

  flag := true;
  other := false;
  flag := flag and other;
  other := flag or other;

  flag := i == j;
  flag := i <> j;
  flag := i < j;
  flag := r > s;
  flag := ch1 <= ch2;
  flag := text1 >= text2;

  i := r + s;          { ERROR: integer := real }
  i := i / j;          { ERROR: / returns real }
  r := flag + i;       { ERROR: boolean + integer }
  i := r div j;        { ERROR: div requires integer operands }
  k := i mod s;        { ERROR: mod requires integer operands }
  flag := i and j;     { ERROR: and requires boolean operands }
  other := flag or i;  { ERROR: or requires boolean operands }
  flag := i == flag;   { ERROR: incompatible equality operands }
  flag := flag < other;{ ERROR: ordered comparison cannot use boolean }
end.
